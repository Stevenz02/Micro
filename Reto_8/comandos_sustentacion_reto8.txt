6) CAOS 2 - LATENCIA INDUCIDA (sin editar codigo)
------------------------------------------------------------

docker compose up -d empleados-service           # recrea el contenedor con la variable nueva
.\scripts\generar_trafico.ps1 -SinOnboarding -Consultas 60

#   ~2 minutos despues -> Discord: "Alta Latencia" (empleados-service y api-gateway)

# Volver a la normalidad:
Remove-Item Env:CHAOS_EMPLEADOS_LATENCY_SECONDS
docker compose up -d empleados-service
