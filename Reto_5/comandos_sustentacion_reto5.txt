$GW = "http://localhost:8080"
function Api {
  param([string]$Method, [string]$Path, [string]$Token, $Body)
  $h = @{}
  if ($Token) { $h.Authorization = "Bearer $Token" }
  $p = @{ Method = $Method; Uri = "$GW$Path"; Headers = $h; UseBasicParsing = $true; ContentType = 'application/json; charset=utf-8' }
  if ($null -ne $Body) { $p.Body = [Text.Encoding]::UTF8.GetBytes(($Body | ConvertTo-Json -Depth 5)) }
  try {
    $r = Invoke-WebRequest @p
    $b = if ($r.Content) { $r.Content | ConvertFrom-Json } else { $null }
    [pscustomobject]@{ Status = [int]$r.StatusCode; Body = $b }
  } catch {
    $resp = $_.Exception.Response
    if (-not $resp) { throw }
    $txt = (New-Object IO.StreamReader($resp.GetResponseStream())).ReadToEnd()
    [pscustomobject]@{ Status = [int]$resp.StatusCode; Body = $txt }
  }
}
function Login($email, $pass) { Api POST /auth/login -Body @{ email = $email; password = $pass } }
# Último token SEGURIDAD (activación o recuperación) enviado a un correo. Lo consulta el ADMIN.
function TokenSeguridad($correo) {
  $todas = (Api GET /notificaciones -Token $ADMIN).Body
  $n = $todas | Where-Object { $_.tipo -eq 'SEGURIDAD' -and $_.destinatario -eq $correo } | Select-Object -Last 1
  "Notificacion: $($n.id) -> $($n.mensaje)" | Write-Host
  (Api GET "/notificaciones/seguridad/$($n.id)/token" -Token $ADMIN).Body.token
}

function Esperar($id, $estado) {
  $t = 0
  do {
    Start-Sleep 3; $t += 3
    $e = (Api GET "/vacaciones/$id" -Token $ADMIN).Body.estado
    Write-Host "  [$t s] estado = $e"
  } until ($e -eq $estado -or $t -ge 180)
}
